```javascript
// 排序功能
setSort(e) {
  const field = e.currentTarget.dataset.field;
  const currentSortBy = this.data.sortBy;
  const currentSortOrder = this.data.sortOrder;
  
  if (currentSortBy === field) {
    this.setData({
      sortOrder: currentSortOrder === 'asc' ? 'desc' : 'asc'
    });
  } else {
    this.setData({
      sortBy: field,
      sortOrder: 'desc'
    });
  }
  
  this.setData({
    currentPage: 1
  });
  this.loadData();
}
```

```javascript
// EMG阈值分类
getEmgClass(threshold) {
  if (!threshold) return 'emg-unknown';
  
  const value = parseFloat(threshold);
  if (value >= 80) return 'emg-excellent';
  if (value >= 60) return 'emg-good';
  return 'emg-poor';
},

// 时间格式化
formatTime(timeStr) {
  if (!timeStr) return '未知';
  
  try {
    const date = new Date(timeStr);
    const year = date.getFullYear();
    const month = String(date.getMonth() + 1).padStart(2, '0');
    const day = String(date.getDate()).padStart(2, '0');
    const hours = String(date.getHours()).padStart(2, '0');
    const minutes = String(date.getMinutes()).padStart(2, '0');
    
    return `${year}-${month}-${day} ${hours}:${minutes}`;
  } catch (error) {
    return '未知';
  }
},

// 复制OpenID
copyOpenId(e) {
  const openid = e.currentTarget.dataset.openid;
  wx.setClipboardData({
    data: openid,
    success: () => {
      wx.showToast({
        title: '已复制到剪贴板',
        icon: 'success'
      });
    }
  });
},

  });
```

```javascript
      }
      if (advancedFilters.maxEmg) {
        whereCondition.emgThreshold.$lte = parseFloat(advancedFilters.maxEmg)
      }
    }
    
    if (advancedFilters.quality && advancedFilters.quality !== 'all') {
      switch (advancedFilters.quality) {

          whereCondition.emgThreshold = { $gte: 60, $lt: 80 }
          break
        case 'poor':
          whereCondition.emgThreshold = { $lt: 60 }
          break

    }
    
    // 构建排序条件
    let orderBy = {}
    switch (sortBy) {
      case 'emgThreshold':
        orderBy.emgThreshold = sortOrder === 'asc' ? 1 : -1
        break
      case 'deviceId':
        orderBy.deviceId = sortOrder === 'asc' ? 1 : -1
        break
      case 'recordTime':
      default:
        orderBy.recordTime = sortOrder === 'asc' ? 1 : -1
        break
    }
    
    // 执行查询
    const result = await db.collection('emg_records')
      .where(whereCondition)
      .orderBy(Object.keys(orderBy)[0], Object.values(orderBy)[0])
      .skip((currentPage - 1) * pageSize)
      .limit(pageSize)
      .get()
    
    // 获取统计数据
    const totalResult = await db.collection('emg_records').count()
    const todayResult = await db.collection('emg_records')
      .where({
        recordTime: {
          $gte: new Date(new Date().setHours(0, 0, 0, 0)),
          $lt: new Date(new Date().setHours(23, 59, 59, 999))
        }
      })
      .count()
    const abnormalResult = await db.collection('emg_records')
      .where({ emgThreshold: { $lt: 60 } })
      .count()
    const deviceResult = await db.collection('emg_records')
      .aggregate()
      .group({
        _id: '$deviceId'
      })
      .end()
    
    return {
```

```javascript
// 云函数入口文件
const cloud = require('wx-server-sdk')

cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })

const db = cloud.database()

exports.main = async (event, context) => {
  const { recordId } = event

  if (!recordId) {
    return {
      success: false,
      message: '记录ID不能为空'
    }
  }

  try {
    // 删除记录
    const result = await db.collection('emg_records')
      .doc(recordId)
      .remove()

    if (result.stats.removed > 0) {
      return {
        success: true,
        message: '记录删除成功',
        data: result
      }
    } else {
      return {
        success: false,
        message: '记录不存在或删除失败'
      }
    }
  } catch (error) {
    console.error('删除记录失败:', error)
    return {
      success: false,
      message: '删除记录失败',
      error: error.message
    }
  }
}
```

```javascript
      const endTime = Date.now();

    }
  }
};
```
